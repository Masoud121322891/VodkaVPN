#!/usr/bin/env bash # Publishes the Hiddify user page to GitHub Pages and prints the URL for the panel. # Usage: bash publish.sh USER/REPO [path/to/userpage-pgclock.html] set -euo pipefail REPO="${1:-}"; FILE="${2:-userpage-pgclock.html}" [[ "$REPO" == */* ]] || { echo "Usage: bash publish.sh USER/REPO [file.html]"; exit 1; } [[ -f "$FILE" ]] || { echo "File not found: $FILE"; exit 1; } command -v git >/dev/null || { echo "Install git first."; exit 1; } command -v gh >/dev/null || { echo "Install GitHub CLI (https://cli.github.com), then run: gh auth login"; exit 1; } gh auth status >/dev/null 2>&1 || { echo "Not logged in. Run: gh auth login"; exit 1; } SRC="$(cd "$(dirname "$FILE")" && pwd)/$(basename "$FILE")" TMP="$(mktemp -d)"; trap 'rm -rf "$TMP"' EXIT if gh repo view "$REPO" >/dev/null 2>&1; then gh repo clone "$REPO" "$TMP/repo" -- -q else gh repo create "$REPO" --public >/dev/null git init -q -b main "$TMP/repo" git -C "$TMP/repo" remote add origin "https://github.com/$REPO.git" fi cp "$SRC" "$TMP/repo/index.html" cd "$TMP/repo" git add index.html git -c user.name="userpage" -c user.email="userpage@users.noreply.github.com" commit -q -m "Update user page" || echo "No changes to commit" git branch -M main git push -q -u origin main gh api -X POST "repos/$REPO/pages" -f "source[branch]=main" -f "source[path]=/" >/dev/null 2>&1 || true echo echo "Done. In the Hiddify panel go to Settings > Customizing User Page > User page URL and paste:" echo " https://${REPO%%/*}.github.io/${REPO##*/}/" echo "(GitHub Pages can take about a minute to go live.)"
